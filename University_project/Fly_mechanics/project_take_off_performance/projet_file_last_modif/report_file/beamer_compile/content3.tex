	\section{Perspectives et axes d'amélioration}
	
	\begin{frame}{Perspectives et axes d'amélioration}
		\begin{itemize}
			\item \textbf{Optimisation des méthodes de calcul:}
			\begin{itemize}
				\item Développement d'une méthode numérique couplée à une \textbf{optimisation sous contraintes}.
			\end{itemize}
			\item \textbf{Enrichissement des paramètres physiques:}
			\begin{itemize}
				\item Intégration dynamique des \textbf{effets thermodynamiques} (pression, densité, température variables).
				\item Prise en compte fine de l'état et de la \textbf{pente de la piste}.
			\end{itemize}
			\item \textbf{Intégration de l'Intelligence Artificielle:}
			\begin{itemize}
				\item Création d'un modèle d'apprentissage capable de prédire les distances selon les contraintes environnementales.
				\item Développement d'un \textbf{agent spécialisé} pour l'estimation automatique des performances.
			\end{itemize}
			\item \textbf{Approche Multi-modèles:}
			\begin{itemize}
				\item Utilisation d'une moyenne pondérée des modèles actuels pour obtenir une prédiction plus fidèle à la réalité opérationnelle.
			\end{itemize}
		\end{itemize}
	\end{frame}
	
	
	
	
	
	
	\section{Conclusion}
	
	\begin{frame}{Conclusion}
		\begin{itemize}
			\item \textbf{Objectif atteint :} Modélisation complète de la phase de décollage, du lâcher des freins au franchissement de l'obstacle, via la confrontation de quatre approches mathématiques.
			\item \textbf{Validation numérique :} Développement d'un programme Python testé sur une flotte variée (Cessna 172, A320, Beluga XL, Rafale), confirmant une excellente cohérence des modèles pour le roulage au sol ($S_a$) .
			\item \textbf{Esprit critique :} Identification des limites des modèles simplifiés pour la phase de montée ($S_c$) et analyse de l'influence prépondérante du rapport poussée/poids.
			\item \textbf{Synthèse :} Ce travail a permis de renforcer les compétences en physique du vol et en informatique, posant une base solide pour de futurs outils de simulation plus complexes .
		\end{itemize}
		
	\end{frame}
