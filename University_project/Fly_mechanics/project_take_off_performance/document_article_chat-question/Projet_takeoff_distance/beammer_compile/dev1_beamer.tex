















	\subsection{Bilan des forces}

\begin{frame}{Forces appliquées à l'avion lors du décollage}
	

	
	\begin{figure}[H]
		\centering
		\includegraphics[width=0.7\linewidth]{billanforce.png}
		\label{fig:placeholder}
		\caption{Schéma technique d'un avion en piste }
	\end{figure}
	

	
	\begin{itemize}
		\item $T$ : Poussée du moteur
		\item $D$ : Traînée aérodynamique
		\item $L$ : Portance
		\item $W$ : Poids de l'avion
		\item $F_f=\mu(W-L)$ : Force de frottement
	\end{itemize}
	
\end{frame}



\begin{frame}{ Bilan de forces }
	
	En appliquant la seconde loi de Newton à l’avion pendant sa phase de roulage au sol, on obtient :	

	\begin{equation} \label{PFD}
		\frac{\vec{W}}{g}\frac{dV}{dt}=\vec{T} - \vec{D} - \vec{F_r} 
	\end{equation}
	


	
\end{frame}
