	\subsection*{}
	\textcolor{blue}{Reponse 3: Application de la loi des gaz parfaits pour un point en phase gazeuse }
	
		
		
		On choisit un point sur la branche \textbf{vapeur sèche} de l'isotherme $T = 25^\circ$C :
		
		\begin{itemize}
			\item $P = 12,6 \, \text{bar} = 1,26 \times 10^6 \, \text{Pa}$
			\item $V = 3,0 \, \text{cm}^3 = 3,0 \times 10^{-6} \, \text{m}^3$
			\item $T = 25 + 273.15 = 298.15 \, \text{K}$
		\end{itemize}
		
		La loi des gaz parfaits s'écrit :
		\begin{equation}
			PV = nRT
		\end{equation}
		
		On en déduit le nombre de moles :
		\begin{align}
			n &= \frac{PV}{RT} \\
			n &= 1,525 \times 10^{-3} \, \text{mol}
		\end{align}
		
	
	
	\subsection*{}	
	\textcolor{blue}{Reponse 4: Enthalpie de vaporisation à $T = 22,5^\circ$C}
		
		On utilise la relation de Clapeyron :
		\begin{equation}
			\frac{dP_\text{sat}}{dT} = \frac{\Delta H_\text{vap}}{T (V_g - V_l)}
		\end{equation}
		
		Entre $T_1 = 20^\circ$C et $T_2 = 25^\circ$C (isothermes expérimentales) :
		\[
		\Delta H_\text{vap} \approx \frac{\Delta P_\text{sat}}{\Delta T} \times T_m \times (V_g - V_l)
		\]
		
		\textbf{Données :}
		\begin{itemize}
			\item $\Delta T = 5\,\text{K}$
			\item $T_m = 22,5 + 273 = 295,5\,\text{K}$
			\item $V_g - V_l \approx 0,4\,\text{cm}^3 = 4\times10^{-7}\,\text{m}^3$
			\item $\Delta P_\text{sat} \approx 0,8\,\text{bar} = 8\times10^4\,\text{Pa}$
		\end{itemize}
		
		\textbf{Calcul :}
		\begin{align}
			\Delta H_\text{vap} &= \frac{8\times10^4}{5} \times 295,5 \times 4\times10^{-7} \\
			&= 1,89\times10^3 \,\text{J/mol} = 59\,\text{kJ/mol}
		\end{align}
		
		\textbf{Enthalpie massique :}
		\[
		\Delta h_\text{vap} = \frac{59\times10^3}{200,59} = 294\,\text{kJ/kg}
		\]
		
		
		
		
	\subsection*{}
	\textcolor{blue}{Reponse 5: Rapport $\frac{P_c V_{mc}}{RT_c}$ (SF$_6$)}
		
		\textbf{Données SF$_6$}
		\begin{itemize}
			\item $T_c = 45,5^\circ$C $= 318,65\,\text{K}$
			\item $P_c = 37,6\,\text{bar} = 3,76\times10^6\,\text{Pa}$
			\item $R = 8,314\,\text{J/mol·K}$
		\end{itemize}
		
		\textbf{Théorie Van der Waals}
		\[
		V_{mc} = 3b, \quad b = \frac{RT_c}{8P_c}
		\]
		
		\[
		\frac{P_c V_{mc}}{RT_c} = \frac{P_c \cdot 3b}{RT_c} = \frac{3}{8} = 0,375
		\]
		
		\textbf{Valeur expérimentale}
		\[
		V_{mc} \approx 0,32\,\text{L/mol} = 3,2\times10^{-4}\,\text{m}^3\text{mol}^{-1}
		\]
		\[
		\frac{P_c V_{mc}}{RT_c} = \frac{3,76\times10^6 \times 3,2\times10^{-4}}{8,314 \times 318,65} = 0,37
		\]
		
		\textbf{Comparaison : $0,37$ vs $0,375$ (accord excellent !)}
		
		
		
		
	\subsection*{}	
	\textcolor{blue}{Reponse 6:  Constantes de l'équation de Van der Waals}
		
		\textbf{Expressions théoriques}
		\begin{align}
			a &= \frac{27 R^2 T_c^2}{64 P_c} \\
			b &= \frac{R T_c}{8 P_c}
		\end{align}
		
		\textbf{Calcul pour SF$_6$}
		\begin{align}
			b &= \frac{8,314 \times 318,65}{8 \times 3,76\times10^6} = 8,82\times10^{-5} \, \text{m}^3\text{mol}^{-1} \\
			a &= \frac{27 \times (8,314)^2 \times (318,65)^2}{64 \times 3,76\times10^6} = 7,15 \, \text{Pa·m}^6\text{mol}^{-2}
		\end{align}
		
		\textbf{Unités SI}
		\begin{itemize}
			\item $a$ : $\text{Pa·m}^6\text{mol}^{-2}$ (ou $\text{N·m}^4\text{mol}^{-2}$)
			\item $b$ : $\text{m}^3\text{mol}^{-1}$
		\end{itemize}
		
		\begin{center}
			\fbox{\begin{minipage}{0.9\textwidth}
					\textbf{Résumé des résultats :}
					
					\begin{itemize}
						\item $\Delta H_\text{vap} = 59\,\text{kJ/mol} = 294\,\text{kJ/kg}$
						\item $\frac{P_c V_{mc}}{RT_c} = 0,37$ (vs $0,375$ théorique)
						\item $a = 7,15\,\text{Pa·m}^6\text{mol}^{-2}$
						\item $b = 8,82\times10^{-5}\,\text{m}^3\text{mol}^{-1}$
					\end{itemize}
			\end{minipage}}
		\end{center}

	if (n == 1) {
		return 1;
	} else {
		return n * factorial(n - 1) ;
